/**
 * Interface defining contract for order processing entities.
 */
interface Orderable {
    double calculateTotal();
    String generateReceipt();
    boolean validateOrder();
    void applyDiscount(double percent);
}

/**
 * Abstract Strategy Class for Discount Strategies (Polymorphism & Abstraction).
 */
abstract class DiscountStrategy {
    private String strategyName;

    public DiscountStrategy(String strategyName) {
        this.strategyName = strategyName;
    }

    public String getStrategyName() { return strategyName; }
    public abstract double calculateDiscount(double rawAmount);
}

/**
 * Concrete Discount Strategies
 */
class PercentageDiscount extends DiscountStrategy {
    private double percentage;

    public PercentageDiscount(String name, double percentage) {
        super(name);
        this.percentage = percentage;
    }

    @Override
    public double calculateDiscount(double rawAmount) {
        return rawAmount * (percentage / 100.0);
    }
}

class VIPDiscount extends DiscountStrategy {
    public VIPDiscount() {
        super("VIP Tier Discount ($15 off over $50)");
    }

    @Override
    public double calculateDiscount(double rawAmount) {
        if (rawAmount >= 50.0) {
            return 15.0;
        }
        return rawAmount * 0.05; // 5% fallback
    }
}

/**
 * Abstract Base Class for Menu Items (Abstraction & Encapsulation).
 */
abstract class MenuItem {
    private String id;
    private String name;
    private double basePrice;
    private String category;

    public MenuItem(String id, String name, double basePrice, String category) {
        this.id = id;
        this.name = name;
        setBasePrice(basePrice);
        this.category = category;
    }

    // Encapsulation: Getters and Setters with data validation
    public String getId() { return id; }
    public void setId(String id) { this.id = id; }

    public String getName() { return name; }
    public void setName(String name) { this.name = name; }

    public double getBasePrice() { return basePrice; }
    public void setBasePrice(double basePrice) {
        if (basePrice < 0) {
            throw new IllegalArgumentException("Base price cannot be negative.");
        }
        this.basePrice = basePrice;
    }

    public String getCategory() { return category; }

    // Abstract Methods
    public abstract double calculateTax();
    public abstract String getItemDetails();
}

// 2. INHERITANCE & POLYMORPHISM: Concrete Menu Subclasses

class FoodItem extends MenuItem {
    private boolean isVegetarian;
    private String prepTime;

    public FoodItem(String id, String name, double basePrice, boolean isVegetarian, String prepTime) {
        super(id, name, basePrice, "Food");
        this.isVegetarian = isVegetarian;
        this.prepTime = prepTime;
    }

    public boolean isVegetarian() { return isVegetarian; }
    public String getPrepTime() { return prepTime; }

    @Override
    public double calculateTax() {
        return getBasePrice() * 0.08; // 8% food tax
    }

    @Override
    public String getItemDetails() {
        return (isVegetarian ? "[Veg] " : "[Non-Veg] ") + "Prep: " + prepTime;
    }
}

class BeverageItem extends MenuItem {
    private String size; // Small, Medium, Large
    private boolean isIceCold;

    public BeverageItem(String id, String name, double basePrice, String size, boolean isIceCold) {
        super(id, name, basePrice, "Beverage");
        this.size = size;
        this.isIceCold = isIceCold;
    }

    public String getSize() { return size; }

    @Override
    public double calculateTax() {
        return getBasePrice() * 0.10; // 10% beverage tax
    }

    @Override
    public String getItemDetails() {
        return "Size: " + size + " (" + (isIceCold ? "Iced" : "Hot/Room Temp") + ")";
    }
}

class ComboItem extends MenuItem {
    private List<MenuItem> bundledItems;
    private double comboDiscountRate; // e.g., 0.15 = 15% off sum

    public ComboItem(String id, String name, List<MenuItem> items, double discountRate) {
        super(id, name, calculateComboPrice(items, discountRate), "Combo");
        this.bundledItems = items;
        this.comboDiscountRate = discountRate;
    }

    private static double calculateComboPrice(List<MenuItem> items, double discountRate) {
        double sum = 0;
        for (MenuItem item : items) sum += item.getBasePrice();
        return sum * (1.0 - discountRate);
    }

    @Override
    public double calculateTax() {
        return getBasePrice() * 0.07; // 7% special combo tax
    }

    @Override
    public String getItemDetails() {
        return "Bundle (" + bundledItems.size() + " items, " + (int)(comboDiscountRate * 100) + "% off)";
    }
}

// 3. ENCAPSULATED ORDER AGGREGATE
// ============================================================================

enum OrderStatus {
    PENDING, PREPARING, COMPLETED, CANCELLED
}

class OrderItem {
    private MenuItem item;
    private int quantity;

    public OrderItem(MenuItem item, int quantity) {
        this.item = item;
        setQuantity(quantity);
    }

    public MenuItem getItem() { return item; }
    public int getQuantity() { return quantity; }
    public void setQuantity(int quantity) {
        if (quantity <= 0) throw new IllegalArgumentException("Quantity must be positive.");
        this.quantity = quantity;
    }

    public double calculateSubtotal() {
        return item.getBasePrice() * quantity;
    }

    public double calculateTaxTotal() {
        return item.calculateTax() * quantity;
    }
}

class Order implements Orderable {
    private String orderId;
    private String customerName;
    private List<OrderItem> items;
    private OrderStatus status;
    private DiscountStrategy discountStrategy;

    public Order(String orderId, String customerName) {
        this.orderId = orderId;
        this.customerName = customerName;
        this.items = new ArrayList<>();
        this.status = OrderStatus.PENDING;
        this.discountStrategy = null;
    }

    public String getOrderId() { return orderId; }
    public String getCustomerName() { return customerName; }
    public OrderStatus getStatus() { return status; }
    public void setStatus(OrderStatus status) { this.status = status; }

    public void setDiscountStrategy(DiscountStrategy strategy) {
        this.discountStrategy = strategy;
    }

    public void addItem(MenuItem item, int qty) {
        for (OrderItem oi : items) {
            if (oi.getItem().getId().equalsIgnoreCase(item.getId())) {
                oi.setQuantity(oi.getQuantity() + qty);
                return;
            }
        }
        items.add(new OrderItem(item, qty));
    }

    public List<OrderItem> getItems() { return items; }

    public double calculateSubtotal() {
        double total = 0;
        for (OrderItem oi : items) total += oi.calculateSubtotal();
        return total;
    }

    public double calculateTotalTax() {
        double tax = 0;
        for (OrderItem oi : items) tax += oi.calculateTaxTotal();
        return tax;
    }

    public double calculateDiscountAmount() {
        if (discountStrategy == null) return 0.0;
        return discountStrategy.calculateDiscount(calculateSubtotal());
    }

    @Override
    public double calculateTotal() {
        return calculateSubtotal() + calculateTotalTax() - calculateDiscountAmount();
    }

    @Override
    public boolean validateOrder() {
        return !items.isEmpty() && customerName != null && !customerName.trim().isEmpty();
    }

    @Override
    public void applyDiscount(double percent) {
        this.discountStrategy = new PercentageDiscount("Manual " + percent + "%", percent);
    }

    @Override
    public String generateReceipt() {
        StringBuilder sb = new StringBuilder();
        sb.append("=========================================\n");
        sb.append("          RESTAURANT RECEIPT             \n");
        sb.append("Order ID: ").append(orderId).append("\n");
        sb.append("Customer: ").append(customerName).append("\n");
        sb.append("Status:   ").append(status).append("\n");
        sb.append("-----------------------------------------\n");
        for (OrderItem oi : items) {
            sb.append(String.format("%-20s x%-2d  $%6.2f\n", 
                oi.getItem().getName(), oi.getQuantity(), oi.calculateSubtotal()));
        }
        sb.append("-----------------------------------------\n");
        sb.append(String.format("Subtotal:            $%8.2f\n", calculateSubtotal()));
        sb.append(String.format("Tax:                 $%8.2f\n", calculateTotalTax()));
        if (discountStrategy != null) {
            sb.append(String.format("Discount (%s): -$%7.2f\n", 
                discountStrategy.getStrategyName(), calculateDiscountAmount()));
        }
        sb.append(String.format("GRAND TOTAL:         $%8.2f\n", calculateTotal()));
        sb.append("=========================================\n");
        return sb.toString();
    }
}

public class OrderingApp extends JFrame {
    private List<MenuItem> catalog;
    private List<Order> orderHistory;
    private Order activeCart;

    private DefaultTableModel catalogModel;
    private DefaultTableModel ordersModel;
    private JTable ordersTable;
    private JLabel lblCartSubtotal, lblCartTax, lblCartTotal;

    public OrderingApp() {
        catalog = new ArrayList<>();
        orderHistory = new ArrayList<>();
        activeCart = new Order("ORD-001", "Guest");

        seedCatalogData();
        initUI();
    }

    private void seedCatalogData() {
        catalog.add(new FoodItem("FOOD-1", "Gourmet Burger", 12.99, false, "15 mins"));
        catalog.add(new FoodItem("FOOD-2", "Veggie Pasta", 10.50, true, "12 mins"));
        catalog.add(new BeverageItem("BEV-1", "Iced Espresso", 4.50, "Medium", true));
        catalog.add(new BeverageItem("BEV-2", "Green Tea", 3.25, "Large", false));
        
        List<MenuItem> comboSet = new ArrayList<>();
        comboSet.add(catalog.get(0));
        comboSet.add(catalog.get(2));
        catalog.add(new ComboItem("CMB-1", "Burger & Iced Coffee Meal", comboSet, 0.15));
    }

    private void initUI() {
        setTitle("Java OOP Restaurant Ordering System");
        setSize(950, 650);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);
        setLayout(new BorderLayout(10, 10));

        // Header
        JPanel header = new JPanel();
        header.setBackground(new Color(0, 115, 150));
        JLabel title = new JLabel("RESTAURANT ORDERING SYSTEM (Java OOP Project)");
        title.setFont(new Font("SansSerif", Font.BOLD, 18));
        title.setForeground(Color.WHITE);
        header.add(title);
        add(header, BorderLayout.NORTH);

        // Main Center Panel
        JSplitPane mainSplit = new JSplitPane(JSplitPane.VERTICAL_SPLIT);
        mainSplit.setDividerLocation(360);

        // Top Split: Catalog & Cart
        JSplitPane topSplit = new JSplitPane(JSplitPane.HORIZONTAL_SPLIT);
        topSplit.setDividerLocation(520);

        // Catalog Panel
        JPanel catalogPanel = new JPanel(new BorderLayout(5, 5));
        catalogPanel.setBorder(BorderFactory.createTitledBorder("Menu Catalog (Polymorphic Items)"));
        
        String[] cols = {"ID", "Name", "Category", "Price ($)", "Details"};
        catalogModel = new DefaultTableModel(cols, 0);
        JTable catalogTable = new JTable(catalogModel);
        refreshCatalogTable();

        catalogPanel.add(new JScrollPane(catalogTable), BorderLayout.CENTER);

        JButton btnAddToCart = new JButton("Add Selected to Cart");
        btnAddToCart.addActionListener(e -> {
            int row = catalogTable.getSelectedRow();
            if (row >= 0) {
                String id = (String) catalogModel.getValueAt(row, 0);
                MenuItem item = findCatalogItem(id);
                if (item != null) {
                    activeCart.addItem(item, 1);
                    updateCartUI();
                }
            } else {
                JOptionPane.showMessageDialog(this, "Select a item from catalog first!");
            }
        });
        catalogPanel.add(btnAddToCart, BorderLayout.SOUTH);

        topSplit.setLeftComponent(catalogPanel);

        // Cart Panel
        JPanel cartPanel = new JPanel(new BorderLayout(5, 5));
        cartPanel.setBorder(BorderFactory.createTitledBorder("Active Cart"));
        
        lblCartSubtotal = new JLabel("Subtotal: $0.00");
        lblCartTax = new JLabel("Tax: $0.00");
        lblCartTotal = new JLabel("Total: $0.00");

        JPanel cartTotalsPanel = new JPanel(new GridLayout(3, 1));
        cartTotalsPanel.add(lblCartSubtotal);
        cartTotalsPanel.add(lblCartTax);
        cartTotalsPanel.add(lblCartTotal);

        cartPanel.add(cartTotalsPanel, BorderLayout.CENTER);

        JButton btnSubmitOrder = new JButton("Submit & Process Order");
        btnSubmitOrder.addActionListener(e -> {
            if (!activeCart.validateOrder()) {
                JOptionPane.showMessageDialog(this, "Cart is empty! Add items before checkout.");
                return;
            }
            orderHistory.add(activeCart);
            refreshOrdersTable();
            JOptionPane.showMessageDialog(this, "Order Submitted!\n" + activeCart.generateReceipt());
            
            // Reset Cart
            activeCart = new Order("ORD-00" + (orderHistory.size() + 1), "Customer " + (orderHistory.size() + 1));
            updateCartUI();
        });
        cartPanel.add(btnSubmitOrder, BorderLayout.SOUTH);

        topSplit.setRightComponent(cartPanel);
        mainSplit.setTopComponent(topSplit);

        // Bottom Split: Order History Table
        JPanel ordersPanel = new JPanel(new BorderLayout(5, 5));
        ordersPanel.setBorder(BorderFactory.createTitledBorder("Order Management & History"));

        String[] orderCols = {"Order ID", "Customer", "Subtotal ($)", "Total Tax ($)", "Grand Total ($)", "Status"};
        ordersModel = new DefaultTableModel(orderCols, 0);
        ordersTable = new JTable(ordersModel);

        ordersPanel.add(new JScrollPane(ordersTable), BorderLayout.CENTER);
        mainSplit.setBottomComponent(ordersPanel);

        add(mainSplit, BorderLayout.CENTER);
    }

    private MenuItem findCatalogItem(String id) {
        for (MenuItem item : catalog) {
            if (item.getId().equalsIgnoreCase(id)) return item;
        }
        return null;
    }

    private void refreshCatalogTable() {
        catalogModel.setRowCount(0);
        for (MenuItem item : catalog) {
            catalogModel.addRow(new Object[]{
                item.getId(),
                item.getName(),
                item.getCategory(),
                String.format("%.2f", item.getBasePrice()),
                item.getItemDetails()
            });
        }
    }

    private void updateCartUI() {
        lblCartSubtotal.setText(String.format("Subtotal: $%.2f", activeCart.calculateSubtotal()));
        lblCartTax.setText(String.format("Tax: $%.2f", activeCart.calculateTotalTax()));
        lblCartTotal.setText(String.format("Total: $%.2f", activeCart.calculateTotal()));
    }

    private void refreshOrdersTable() {
        ordersModel.setRowCount(0);
        for (Order o : orderHistory) {
            ordersModel.addRow(new Object[]{
                o.getOrderId(),
                o.getCustomerName(),
                String.format("%.2f", o.calculateSubtotal()),
                String.format("%.2f", o.calculateTotalTax()),
                String.format("%.2f", o.calculateTotal()),
                o.getStatus()
            });
        }
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            new OrderingApp().setVisible(true);
        });
    }
}
